process PurchaseApproval display='Purchase approval' package=Purchasing use=orchestration subject=PurchaseRequest bound=- enum=- initial=Drafting
  keys e:<PurchaseRequest> e:<PurchaseApproval> s:inflection
  context 0 amount: builtin decimal default=0
  context 1 reminders: builtin int32 default=0
  event submit actors=Requester,BudgetHolder payload=- transitions=1
  event requestChanges actors=Requester,BudgetHolder payload=- transitions=1
  event approve actors=Approver,FinanceController payload=- transitions=1
  event reject actors=Approver,FinanceController payload=reason:text! transitions=1
  action sendReminder expr=({ reminders: context.reminders + 1 }) raises=- used-by=t(after 5d / sendReminder)
  roots Drafting,Review,Approval,Ordering,Ordered,Rejected atomic=Drafting,Review.Budget.Checking,Review.Budget.BudgetOk,Review.Budget.BudgetRejected,Review.Compliance.Pending,Review.Compliance.Cleared,Approval,Ordering,Ordered,Rejected bound=Drafting,Review,Approval,Ordering,Ordered,Rejected
  state Drafting atomic depth=1 parent=- children=- leaf out=1
  state Review parallel depth=1 parent=- children=Budget,Compliance out=1
  state   Review.Budget compound depth=2 parent=Review children=Checking,BudgetOk,BudgetRejected initial=Checking region=0 out=0
  state     Review.Budget.Checking atomic depth=3 parent=Review.Budget children=- leaf invoke=checkBudget out=2
  state     Review.Budget.BudgetOk final depth=3 parent=Review.Budget children=- final leaf out=0
  state     Review.Budget.BudgetRejected atomic depth=3 parent=Review.Budget children=- leaf out=1
  state   Review.Compliance compound depth=2 parent=Review children=Pending,Cleared initial=Pending region=1 out=0
  state     Review.Compliance.Pending atomic depth=3 parent=Review.Compliance children=- leaf invoke=complianceReview out=1
  state     Review.Compliance.Cleared final depth=3 parent=Review.Compliance children=- final leaf out=0
  state Approval atomic depth=1 parent=- children=- leaf out=3
  state Ordering atomic depth=1 parent=- children=- leaf invoke=createPurchaseOrder out=1
  state Ordered final depth=1 parent=- children=- final leaf out=0
  state Rejected final depth=1 parent=- children=- final leaf out=0
  transition 'submit' Drafting -> Review trigger=event event=submit
  transition 'done: checkBudget' Review.Budget.Checking -> Review.Budget.BudgetOk trigger=invoke-done invoke=checkBudget
  transition 'error: checkBudget' Review.Budget.Checking -> Review.Budget.BudgetRejected trigger=invoke-error invoke=checkBudget
  transition 'requestChanges' Review.Budget.BudgetRejected -> Drafting trigger=event event=requestChanges
  transition 'done: complianceReview' Review.Compliance.Pending -> Review.Compliance.Cleared trigger=invoke-done invoke=complianceReview
  transition 'done' Review -> Approval trigger=done
  transition 'approve' Approval -> Ordering trigger=event event=approve gate=purchaseApproval
  transition 'reject' Approval -> Rejected trigger=event event=reject
  transition 'after 5d / sendReminder' Approval -> (targetless) trigger=after after=P5D ms=432000000 actions=sendReminder
  transition 'done: createPurchaseOrder' Ordering -> Ordered trigger=invoke-done invoke=createPurchaseOrder
  gate purchaseApproval on 'approve' required=2 signers=Approver,FinanceController required-actors=FinanceController repeat=false reason=false meanings=approved('Approved for purchase')
    audit instance:string! process:id! gate:id! transition:id! sequence:int64! signer:string actor:id meaning:id reason:text at:datetimeoffset! outcome:string![signed|completed|refused|discarded] costCentre:string(attribute)
  invoke checkBudget service state=Review.Budget.Checking process=- actors=-
  invoke complianceReview human-task state=Review.Compliance.Pending process=- actors=ComplianceOfficer
  invoke createPurchaseOrder service state=Ordering process=- actors=ProcurementSystem
  actors Approver,BudgetHolder,ComplianceOfficer,FinanceController,ProcurementSystem,Requester membership e:<PurchaseRequest> e:<PurchaseApproval> k:actor
  scenarios BudgetRejected,ChangesRequested,ComplianceFirstThenBudget,HappyPath,RejectionWithReason,ReminderAfterFiveDays membership k:scenario r:<PurchaseApproval>
